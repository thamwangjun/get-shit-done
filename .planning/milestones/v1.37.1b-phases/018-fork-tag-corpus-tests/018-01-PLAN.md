'use strict';

/**
 * Fork Persona Tag Tests
 *
 * Regression guard for the <persona> → <role> upstream revert risk.
 * All 31 agents in agents/gsd-*.md must use <persona> tag, not <role>.
 * Upstream agents use <role>; every upstream merge that touches an agent
 * file risks reverting this fork change silently.
 *
 * Code-fence content is excluded from the <role> absence check to allow
 * agent files to document the upstream <role> tag in prose examples.
 */

const { test, describe } = require('node:test');
const assert = require('node:assert/strict');
const fs = require('fs');
const path = require('path');

const AGENTS_DIR = path.join(__dirname, '..', 'agents');

const agents = fs.readdirSync(AGENTS_DIR)
  .filter(f => f.startsWith('gsd-') && f.endsWith('.md'));

describe('PERSONA: agent files use <persona>, not <role>', () => {
  for (const file of agents) {
    test(`${file} contains <persona> tag`, () => {
      const content = fs.readFileSync(path.join(AGENTS_DIR, file), 'utf-8');
      assert.ok(
        content.includes('<persona>'),
        `${file} must contain <persona> tag — upstream merge may have reverted to <role>`
      );
    });

    test(`${file} does not use <role> as persona XML tag`, () => {
      const content = fs.readFileSync(path.join(AGENTS_DIR, file), 'utf-8');
      // Strip code fence content to allow <role> in documentation examples
      const withoutFences = content.replace(/```[\s\S]*?```/g, '');
      assert.ok(
        !/<role>/.test(withoutFences),
        `${file} must not use <role> as persona XML tag (outside code fences) — use <persona> instead`
      );
    });
  }
});
```

Key points:
- File uses `agents` (not `ALL_AGENTS`) as the variable name — keeps filenames with .md extension for readable test names
- 2 subtests per agent (62 total for 31 agents): one <persona> presence, one <role> absence
- The <role> absence check uses regex on withoutFences (not raw content) — prevents false positives from code examples
- readFileSync is called inside each test() callback, not pre-cached at module load
- describe label uses 'PERSONA: ' prefix per convention

'use strict';

/**
 * Fork Intent Tag Tests
 *
 * Regression guard for the <task>/<objective> → <intent> rename in the command layer.
 * The fork uses <intent> as the primary directive block in command files.
 * Upstream commands use <task> or <objective> for the same role.
 *
 * DESIGN NOTE — Expected failures on current corpus:
 * This test enforces the correct standard. As of 2026-04-28, 32 command files
 * still use <objective> (older upstream convention) and will fail their subtest.
 * These failures are BY DESIGN — they document the gap between the corpus state
 * and the fork standard. TEST-GATE-01 success requires only that no NEW failures
 * are introduced in pre-existing tests (not that this test passes 100%).
 * The <objective> → <intent> conversion is deferred to a follow-on phase.
 *
 * What this guards:
 * - Upstream adding new command files with bare <task> or <objective> blocks
 * - Upstream merges reverting existing <intent> blocks back to <task>/<objective>
 *
 * Bare block detection: a line where line.trim() === '<task>' or '<objective>'
 * (not prose mentions, attribute values, or code examples).
 */

const { test, describe } = require('node:test');
const assert = require('node:assert/strict');
const fs = require('fs');
const path = require('path');

const COMMANDS_DIR = path.join(__dirname, '..', 'commands', 'gsd');

const commands = fs.readdirSync(COMMANDS_DIR)
  .filter(f => f.endsWith('.md'))
  .sort();

describe('INTENT: command files use <intent>, not <task> or <objective>', () => {
  for (const file of commands) {
    test(`${file} does not use bare <task> or <objective> as primary directive block`, () => {
      const content = fs.readFileSync(path.join(COMMANDS_DIR, file), 'utf-8');
      const lines = content.split('\n');
      const violations = [];
      for (let i = 0; i < lines.length; i++) {
        const trimmed = lines[i].trim();
        if (trimmed === '<task>') {
          violations.push(`line ${i + 1}: bare <task> block — replace with <intent>`);
        }
        if (trimmed === '<objective>') {
          violations.push(`line ${i + 1}: bare <objective> block — replace with <intent>`);
        }
      }
      assert.deepEqual(
        violations,
        [],
        `${file} uses bare directive block(s) that should be <intent>:\n${violations.join('\n')}`
      );
    });
  }
});
```

Key points:
- One subtest per command file (79 subtests total)
- Per D-01 (locked decision): guards against BOTH <objective> AND <task> as bare outermost blocks — stricter than current corpus
- 47 files with <intent> will pass; 32 files with <objective> will fail (expected, by design, documented in the file header)
- Detection uses `trimmed === '<task>'` and `trimmed === '<objective>'` — strict equality on trimmed lines, not substring match
- Violations array per file gives precise line numbers in failure output
- assert.deepEqual(violations, [], ...) form: empty array means no violations
- Test file header documents that 32 failures are expected and by design (TEST-GATE-01 does not require this test to be 100% green)
